第一阶段：系统准备与基础软件安装

macOS

bash
# 1. 安装 Xcode Command Line Tools
xcode-select --install

# 2. 安装 Homebrew
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

# 3. 安装所有必要软件包
brew install neovim starship tmux lazygit fzf zoxide direnv thefuck bat ripgrep fd tokei sd git-delta llvm nasm fish

# 4. 安装字体与应用程序
brew install --cask ghostty font-lxgw-wenkai-gb font-jetbrains-mono-nerd-font obsidian zed

# 5. 安装 C3 编译器
brew install c3c

# 6. 安装专业开发附加工具（可选）
brew install gdb lldb hyperfine valgrind cmake ninja ccache qemu bear cppcheck
brew install --cask renderdoc blender

# 7. 安装 Zellij
brew install zellij

# 8. 安装 Haskell 工具链（ghcup）
curl --proto '=https' --tlsv1.2 -sSf https://get-ghcup.haskell.org | sh

# 9. 安装 Emacs Plus（macOS 优化版，含原生编译）
brew tap d12frosted/emacs-plus
brew trust --formula d12frosted/emacs-plus/emacs-plus@30
brew install d12frosted/emacs-plus/emacs-plus@30
ln -sf /opt/homebrew/Cellar/emacs-plus@30/30.2/bin/emacs /opt/homebrew/bin/emacs
ln -sf /opt/homebrew/Cellar/emacs-plus@30/30.2/bin/emacsclient /opt/homebrew/bin/emacsclient

# 10. 安装 Doom Emacs
git clone https://github.com/doomemacs/doomemacs ~/.config/emacs
~/.config/emacs/bin/doom install
~/.config/emacs/bin/doom sync

完整 .zshrc 配置

bash
cat > ~/.zshrc << 'EOF'
# If you come from bash you might have to change your $PATH.
# export PATH=$HOME/bin:$HOME/.local/bin:/usr/local/bin:$PATH

# Path to your Oh My Zsh installation.
export ZSH="$HOME/.oh-my-zsh"

# Set name of the theme to load
ZSH_THEME="robbyrussell"

# Which plugins would you like to load?
plugins=(git zsh-syntax-highlighting zsh-autosuggestions zsh-completions)

source $ZSH/oh-my-zsh.sh

# Starship 彩虹提示符
eval "$(starship init zsh)"

# 讓 ls 輸出彩色
export CLICOLOR=1
export LSCOLORS=ExGxBxDxCxEgEdxbxgxcxd

# 語法高亮樣式：無效命令顯示為白色
ZSH_HIGHLIGHT_STYLES[incorrect-command]='fg=white'

# fzf 模糊搜尋
[ -f ~/.fzf.zsh ] && source ~/.fzf.zsh

# zoxide 智能目錄跳轉
eval "$(zoxide init zsh)"

# 歷史記錄優化
setopt HIST_IGNORE_ALL_DUPS
setopt HIST_SAVE_NO_DUPS
setopt SHARE_HISTORY
export HISTSIZE=50000
export SAVEHIST=50000

# direnv 自動加載項目環境變量
eval "$(direnv hook zsh)"

# Lazygit 別名
alias lg='lazygit'

# thefuck 命令糾錯
eval "$(thefuck --alias fix)"

# 遊戲引擎快捷命令
alias cry-vm="open -a 'Parallels Desktop'"

Fish 配置（最终版）

bash
mkdir -p ~/.config/fish
cat > ~/.config/fish/config.fish << 'EOF'
# 基本設置
set -gx LANG en_US.UTF-8
set -gx CLICOLOR 1
set -gx LSCOLORS ExGxBxDxCxEgEdxbxgxcxd

# PATH 設置
# 將 Homebrew LLVM 放在最前，優先使用新編譯器
# 再添加 OpenJDK 的 bin 目錄
set -gx PATH /opt/homebrew/opt/llvm/bin /opt/homebrew/opt/openjdk/bin $HOME/.cargo/bin $PATH

# JAVA_HOME
set -gx JAVA_HOME /opt/homebrew/opt/openjdk/libexec/openjdk.jdk/Contents/Home

# GHCup 環境
if test -f $HOME/.ghcup/env
    bass source $HOME/.ghcup/env
end

# Starship 提示符
starship init fish | source

# fzf 模糊搜尋
if type -q fzf
    fzf --fish | source
end

# zoxide 智能目錄跳轉
if type -q zoxide
    zoxide init fish | source
end

# direnv
if type -q direnv
    direnv hook fish | source
end

# thefuck
if type -q thefuck
    thefuck --alias | source
end

# 歷史記錄優化
set -g fish_history_max_entries 50000
set -g fish_history_ignore duplicates

# Emacs
alias em 'emacsclient -nw -a ""'
function emacs
    /opt/homebrew/bin/emacs --eval '(toggle-frame-maximized)' $argv
end

# 別名
alias lg 'lazygit'
alias godot-gd '/Applications/Godot.app/Contents/MacOS/Godot'
alias godot-cs '/Applications/Godot_mono.app/Contents/MacOS/Godot'

    echo (string replace -r '^'"$HOME" '~' $PWD)

# 其他設置
set -g fish_greeting
EOF

-- 獨立自動啟動 C3 LSP（僅在打開 .c3 文件時生效，不影響其他語言）
local function start_c3_lsp()
  if vim.bo.filetype == "c3" then
    local clients = vim.lsp.get_clients({ bufnr = 0, name = "c3_lsp" })
    if #clients == 0 then
      vim.lsp.start({
        name = "c3_lsp",
        cmd = { vim.fn.stdpath("data") .. "/mason/bin/c3lsp" },
        root_dir = vim.fn.getcwd(),
        filetypes = { "c3" },
      })
    end
  end
end

-- 文件類型變為 c3 時啟動
vim.api.nvim_create_autocmd("FileType", {
  pattern = "c3",
  callback = start_c3_lsp,
})

  callback = start_c3_lsp,

Zed 中配置 Vim 模式：

json
{
    "vim_mode": true

在 ~/.config/zed/keymap.json 中配置 Vim 键位：

json
[
    {
        "context": "Editor && vim_mode == insert",
        "bindings": {
            "j k": ["workspace::SendKeystrokes", "escape"]
        }
    },
    {
        "context": "VimControl && !menu && vim_mode != operator",
        "bindings": {
            "w": "vim::NextSubwordStart",
            "b": "vim::PreviousSubwordStart",
            "e": "vim::NextSubwordEnd",
            "g d": "editor::GoToDefinition",
            "g D": "editor::GoToDeclaration",
            "g y": "editor::GoToTypeDefinition",
            "g I": "editor::GoToImplementation",
            "g A": "editor::FindAllReferences",
            "c d": "editor::Rename",
            "g h": "editor::Hover"
        }
    }
]
Zed 中配置 Emacs 风格键位：

在 ~/.config/zed/keymap.json 中添加：

json
{
    "context": "Editor",
    "bindings": {
        "ctrl-x ctrl-s": "workspace::Save",
        "ctrl-x ctrl-f": "file_finder::Toggle",
        "ctrl-x ctrl-c": "workspace::CloseWindow",
        "alt-x": "command_palette::Toggle",
        "ctrl-s": "workspace::Save"
    }
}

lua
